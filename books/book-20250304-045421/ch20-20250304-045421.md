#### 基本的な使い方

AnkiConnectを活用して、以下の学習方法を実践してみましょう。

1.  **質問応答形式のカード:** 単純な暗記だけでなく、質問応答形式のカードを作成し、能動的に情報を想起する練習を行います。
    *   例: 歴史の年号を暗記するのではなく、「〇〇の変が起こったのは何年？」という質問に答える形式にする。
2.  **キーワードと連想:** キーワードを中心に、関連する情報を連想させるカードを作成し、知識のネットワークを構築します。
    *   例: 「ルネサンス」というキーワードから、「レオナルド・ダ・ヴィンチ」、「フィレンツェ」、「人文主義」などの関連情報を連想させる。

4.  **音声や動画を組み込む:** 発音の練習や、複雑な概念の理解に役立つ音声や動画をカードに組み込みます。
    *   例: 外国語学習で、単語の発音記号と音声ファイルをカードに登録する。

#### 実践的なテクニック

*   **Context Cluesの活用:** 単語の意味を覚える際、例文や関連する画像、音声などをカードに含めることで、文脈の中で単語を理解し、記憶に定着させやすくなります。AnkiConnectを利用して、オンライン辞書や翻訳サービスから例文を自動的に取得し、カードに組み込むことができます。

```python
import requests
import json

def get_context_clues(word):
  """オンライン辞書から例文を取得する関数."""
  url = f"https://api.dictionaryapi.dev/api/v2/entries/en/{word}" # フリーのAPIを利用
  response = requests.get(url)
  data = json.loads(response.text)

  if response.status_code == 200 and data:
    meanings = data[0].get('meanings', [])
    if meanings:
        for meaning in meanings:
            definitions = meaning.get('definitions', [])
            for definition in definitions:
                example = definition.get('example', None)
                if example:
                    return example
  return "例文が見つかりませんでした。"

# AnkiConnectにカードを追加する関数
def add_card_with_context(word):
  example = get_context_clues(word)
  note = {
    "deckName": "English Vocabulary",
    "modelName": "Basic",
    "fields": {
      "Front": word,
      "Back": example
    },
    "options": {
      "allowDuplicate": False,
      "duplicateScope": "deck"
    },
    "tags": ["vocabulary"]
  }

  data = {
    "action": "addNote",
    "version": 6,
    "params": {
      "note": note
    }
  }
  headers = {'Content-Type': 'application/json'}
  response = requests.post('http://localhost:8765', data=json.dumps(data), headers=headers)
  print(response.text)

# 使用例
word = "ubiquitous"
add_card_with_context(word)

```

```python
import requests
from bs4 import BeautifulSoup
import json

def scrape_wikipedia(url):
  """Wikipediaの記事から情報をスクレイピングする関数."""
  response = requests.get(url)
  soup = BeautifulSoup(response.content, "html.parser")
  title = soup.find(id="firstHeading").text
  content = ""
  for paragraph in soup.find(id="bodyContent").find_all("p"):
    content += paragraph.text
  return title, content

def add_card_to_anki(front, back, deck_name="Wikipedia Notes"):
  """Ankiにカードを追加する関数."""
  note = {
    "deckName": deck_name,
    "modelName": "Basic",
    "fields": {
      "Front": front,
      "Back": back
    },
    "options": {
      "allowDuplicate": False,
      "duplicateScope": "deck"
    },
    "tags": ["wikipedia"]
  }

  data = {
    "action": "addNote",
    "version": 6,
    "params": {
      "note": note
    }
  }
  headers = {'Content-Type': 'application/json'}
  response = requests.post('http://localhost:8765', data=json.dumps(data), headers=headers)
  print(response.text)

# 使用例
url = "https://en.wikipedia.org/wiki/Artificial_intelligence"
title, content = scrape_wikipedia(url)
add_card_to_anki(title, content)
```
