4.  **抽出した情報をLINE WORKSに送信:** 抽出した情報をLINE WORKSに送信し、ユーザーに提供します。

以下は、LINE WORKSのメッセージをClaude 3.7に送信し、要約結果をLINE WORKSに返信するPythonコードの例です。

```python
import requests
import json

# LINE WORKS APIの設定
LINE_WORKS_API_URL = "YOUR_LINE_WORKS_API_URL"
LINE_WORKS_BOT_ID = "YOUR_LINE_WORKS_BOT_ID"
LINE_WORKS_ACCESS_TOKEN = "YOUR_LINE_WORKS_ACCESS_TOKEN"

# Claude 3.7 APIの設定
CLAUDE_API_KEY = "YOUR_CLAUDE_API_KEY"
CLAUDE_API_URL = "https://api.anthropic.com/v1/messages"

def send_message_to_line_works(message):
  """LINE WORKSにメッセージを送信する"""
  headers = {
      "Content-Type": "application/json",
      "Authorization": f"Bearer {LINE_WORKS_ACCESS_TOKEN}"
  }
  data = {
      "content": message
  }
  response = requests.post(LINE_WORKS_API_URL, headers=headers, data=json.dumps(data))
  return response

def summarize_with_claude(text):
  """Claude 3.7 APIを使用してテキストを要約する"""
  headers = {
      "Content-Type": "application/json",
      "x-api-key": CLAUDE_API_KEY,
      "anthropic-version": "2023-06-01" # Claudeのバージョン指定
  }
  data = {
      "model": "claude-3-opus-20240229", # Claude 3 Opusを使用
      "max_tokens": 1024,
      "messages": [
          {
              "role": "user",
              "content": f"以下のテキストを要約してください：\n\n{text}"
          }
      ]
  }

  response = requests.post(CLAUDE_API_URL, headers=headers, json=data)
  response_json = response.json()

  # APIエラーハンドリング
  if response.status_code != 200:
      print(f"Claude APIエラー: {response_json}")
      return "要約に失敗しました。"

  try:
      summary = response_json['content'][0]['text']
      return summary
  except KeyError:
      print(f"Claude APIエラー: 応答形式が不正です。: {response_json}")
      return "要約に失敗しました。"


# LINE WORKSからメッセージを受信する（簡略化のため、実際にはAPIで受信する必要があります）
line_works_message = "今日の会議では、新商品の開発戦略について議論しました。市場調査の結果、若年層向けのマーケティング戦略が重要であることがわかりました。また、競合他社の動向についても分析し、差別化戦略を検討する必要があるという結論に至りました。"

# Claude 3.7でメッセージを要約する
summary = summarize_with_claude(line_works_message)

# 要約結果をLINE WORKSに送信する
if summary:
  response = send_message_to_line_works(f"要約結果：\n{summary}")
  if response.status_code == 200:
      print("要約結果をLINE WORKSに送信しました。")
  else:
      print(f"LINE WORKS APIエラー: {response.status_code} - {response.text}")
else:
  print("要約に失敗しました。")
```

**注意点:** 上記のコードはあくまで例であり、実際にはLINE WORKS APIの認証処理やエラーハンドリングなどを適切に実装する必要があります。また、Claude 3.7 APIの利用には、Anthropic社への登録とAPIキーの取得が必要です。Claude 3のモデル指定（上記では`claude-3-opus-20240229`）は、適宜最新のものに変更してください。

**1. LINE WORKSからメッセージを受信する (Webhook):**

LINE WORKSのWebhookを設定し、メッセージを受信する例です。 FlaskなどのWebフレームワークが必要です。

```python
from flask import Flask, request, jsonify
import os
from dotenv import load_dotenv

load_dotenv()  # 環境変数をロード

app = Flask(__name__)

LINE_WORKS_BOT_ID = os.getenv("LINE_WORKS_BOT_ID")
LINE_WORKS_ACCESS_TOKEN = os.getenv("LINE_WORKS_ACCESS_TOKEN")

@app.route('/webhook', methods=['POST'])
def webhook():
    if request.method == 'POST':
        data = request.get_json()
        print("Received webhook data:", data)

        # メッセージの内容を取得
        try:
            message_content = data['content']['text']
            sender_id = data['source']['userId']  # 送信者のuserIdを取得

            # ここでメッセージの内容を処理 (例: Claudeに送信)
            print(f"Received message: {message_content} from user: {sender_id}")
            # 送信者IDも含めてClaudeに送ると、よりパーソナライズされた応答が可能

            #  Claudeにメッセージを送信し、応答を得る関数（前のセクションで定義したものを使用）
            claude_response = summarize_with_claude(message_content) # 例

            #  LINE WORKSに返信する関数（前のセクションで定義したsend_message_to_line_worksを使用）
            send_message_to_line_works(f"応答: {claude_response}")  # 例

        except KeyError as e:
            print(f"KeyError: {e}")
            return jsonify({'status': 'error', 'message': 'Invalid webhook data format'}), 400


        return jsonify({'status': 'success'}), 200
    else:
        return jsonify({'status': 'error', 'message': 'Method not allowed'}), 405

if __name__ == '__main__':
    app.run(debug=True, port=5000)
```

**注意点:** このコードを実行するには、Flaskをインストールする必要があります (`pip install Flask`)。また、LINE WORKSのWebhook設定で、このエンドポイントを登録する必要があります。

**2. LINE WORKSに画像付きメッセージを送信する:**

LINE WORKS APIを使って、画像付きのメッセージを送信する例です。

```python
import requests
import json
import os
from dotenv import load_dotenv

load_dotenv()

LINE_WORKS_API_URL = os.getenv("LINE_WORKS_API_URL")
LINE_WORKS_BOT_ID = os.getenv("LINE_WORKS_BOT_ID")
LINE_WORKS_ACCESS_TOKEN = os.getenv("LINE_WORKS_ACCESS_TOKEN")

def send_image_message_to_line_works(message, image_path):
    """LINE WORKSに画像付きメッセージを送信する"""

    # 画像をアップロード
    upload_url = f"{LINE_WORKS_API_URL}/message/v1/bot/{LINE_WORKS_BOT_ID}/message/image"
    with open(image_path, 'rb') as image_file:
        files = {'file': image_file}
        headers = {
          "Authorization": f"Bearer {LINE_WORKS_ACCESS_TOKEN}"
        }

        upload_response = requests.post(upload_url, files=files, headers=headers)

    if upload_response.status_code != 200:
        print(f"Image upload failed: {upload_response.status_code} - {upload_response.text}")
        return False

    upload_data = upload_response.json()
    image_id = upload_data['imageId']  # アップロードされた画像のID

    # メッセージを送信
    message_url = f"{LINE_WORKS_API_URL}/message/v1/bot/{LINE_WORKS_BOT_ID}/message/push"
    headers = {
        "Content-Type": "application/json",
        "Authorization": f"Bearer {LINE_WORKS_ACCESS_TOKEN}"
    }

    data = {
        "accountId": LINE_WORKS_BOT_ID,  # 送信元のaccountIdを指定
        "content": {
            "type": "text",
            "text": message,
            "image": {
                "imageId": image_id
            }
        }
    }

    response = requests.post(message_url, headers=headers, data=json.dumps(data))

    if response.status_code == 200:
        print("Image message sent successfully!")
        return True
    else:
        print(f"Message send failed: {response.status_code} - {response.text}")
        return False


# 例
image_path = "path/to/your/image.jpg"  # 画像ファイルのパスを指定
message_text = "画像付きメッセージです！"
if send_image_message_to_line_works(message_text, image_path):
    print("Message sent!")
else:
    print("Failed to send message.")
```

**注意点:** このコードを実行するには、LINE WORKS APIの設定に加えて、画像ファイルのパスを正しく設定する必要があります。また、LINE WORKS APIで画像のアップロードが許可されているか確認してください。

**3. Claude 3.7 APIでテキストの分類を行う:**

Claude 3.7 APIを使って、テキストを特定のカテゴリに分類する例です。

```python
import requests
import json
import os
from dotenv import load_dotenv

load_dotenv()

CLAUDE_API_KEY = os.getenv("YOUR_CLAUDE_API_KEY")
CLAUDE_API_URL = "https://api.anthropic.com/v1/messages"

def classify_text_with_claude(text, categories):
    """Claude 3.7 APIを使用してテキストを分類する"""
    headers = {
        "Content-Type": "application/json",
        "x-api-key": CLAUDE_API_KEY,
        "anthropic-version": "2023-06-01"
    }
    data = {
        "model": "claude-3-opus-20240229",
        "max_tokens": 256,
        "messages": [
            {
                "role": "user",
                "content": f"以下のテキストを、{', '.join(categories)} のいずれかのカテゴリに分類してください。\n\nテキスト: {text}\n\nカテゴリ:"
            }
        ]
    }

    response = requests.post(CLAUDE_API_URL, headers=headers, json=data)
    response_json = response.json()

    if response.status_code != 200:
        print(f"Claude APIエラー: {response_json}")
        return "分類に失敗しました。"

    try:
        classification = response_json['content'][0]['text'].strip()
        return classification
    except KeyError:
        print(f"Claude APIエラー: 応答形式が不正です。: {response_json}")
        return "分類に失敗しました。"

# 例
text = "新しいiPhoneの発売日が発表されました。"
categories = ["テクノロジー", "ビジネス", "エンターテイメント"]
classification = classify_text_with_claude(text, categories)

if classification:
    print(f"テキストの分類結果: {classification}")
else:
    print("分類に失敗しました。")
```

**注意点:** このコードを実行するには、Claude 3.7 APIの設定が必要です。また、`categories`に分類したいカテゴリをリスト形式で指定してください。 プロンプトの調整によって、分類精度を向上させることができます。
